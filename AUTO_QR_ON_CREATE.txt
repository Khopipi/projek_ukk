File: app/Models/PengajuanSurat.php

protected static function boot()
{
    parent::boot();

    // STEP 1: Generate token saat creating (sebelum record saved)
    static::creating(function ($model) {
        if (empty($model->nomor_pengajuan)) {
            $model->nomor_pengajuan = self::generateNomorPengajuan();
        }
        
        if (empty($model->signature_token)) {
            $userId = auth()?->id() ?? 1;
            $model->signature_token = QrCodeGenerator::generateSignatureToken(
                $model->id ?? time(),
                $userId
            );
            $model->signature_generated_at = now();
        }
    });

    // STEP 2: Generate QR file setelah record saved (punya ID)
    static::created(function ($model) {
        try {
            if ($model->signature_token) {
                $qrUrl = QrCodeGenerator::generateQrUrl($model->signature_token);
                QrCodeGenerator::generateAndSaveQrCode($qrUrl);
            }
        } catch (\Exception $e) {
            Log::warning('QR generation failed for pengajuan ' . $model->id);
        }
    });
}
